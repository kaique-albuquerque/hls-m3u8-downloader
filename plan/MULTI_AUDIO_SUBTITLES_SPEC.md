# StreamGrab — Multi-Áudio e Legendas (Especificação)

> **Código da feature:** P12
> **Status:** ESPEC — aguardando aprovação para implementação
> **Data:** 2026-08-14

---

## 0. Objetivo

Permitir que o StreamGrab **identifique, selecione e baixe** múltiplas faixas de áudio e legendas quando disponíveis na fonte. Suporte incremental por fonte:

| Fase | Fonte | Multi-áudio | Legendas |
|------|-------|:-----------:|:-------:|
| **P12.1** | YouTube / yt-dlp (redes sociais) | ✅ | ✅ |
| **P12.2** | HLS (mdstrm e outros) | ✅ | ⚠️ |
| **P12.3** | DASH | ✅ | ⚠️ |

> ⚠️ = depende da fonte suportar legendas embutidas na playlist (nem todas suportam)

**Nova função `buildAudioTracks`:**

```js
/**
 * Agrupa formatos de áudio por idioma, ordenados por bitrate desc.
 * Cada idioma gera um AudioTrack com 1+ formatos.
 */
function buildAudioTracks(audioFormats) {
  const byLang = new Map();
  for (const f of audioFormats) {
    const lang = f.language || 'und';
    if (!byLang.has(lang)) {
      byLang.set(lang, {
        language: lang,
        label: f.label || lang,
        isDefault: f.isDefault,
        formats: [],
      });
    }
    byLang.get(lang).formats.push(f);
  }
  // Ordenar: default primeiro, depois por quantidade de formatos
  return [...byLang.values()].sort((a, b) => {
    if (a.isDefault !== b.isDefault) return a.isDefault ? -1 : 1;
    return b.formats.length - a.formats.length;
  });
}
```

**Nova função `buildSubtitleTracks`:**

```js
/**
 * Extrai legendas do JSON do yt-dlp (chaves subtitles + automatic_captions).
 */
function buildSubtitleTracks(info) {
  const tracks = [];
  const seen = new Set();

  // Legendas manuais
  for (const [lang, entries] of Object.entries(info?.subtitles || {})) {
    const best = entries.find((e) => e.ext === 'vtt') || entries[0];
    if (!best) continue;
    const key = `${lang}:manual`;
    if (seen.has(key)) continue;
    seen.add(key);
    tracks.push({
      language: lang,
      label: best.name || lang,
      isAutoGenerated: false,
      isDefault: false,
      format: { ext: best.ext, url: best.url },
    });
  }

  // Legendas automáticas (ASR)
  for (const [lang, entries] of Object.entries(info?.automatic_captions || {})) {
    if (seen.has(`${lang}:manual`)) continue; // manual tem prioridade
    const best = entries.find((e) => e.ext === 'vtt') || entries[0];
    if (!best) continue;
    tracks.push({
      language: lang,
      label: `${lang} (auto)`,
      isAutoGenerated: true,
      isDefault: false,
      format: { ext: best.ext, url: best.url },
    });
  }

  return tracks;
}
```

### 4.1 Parse de `#EXT-X-MEDIA`

**Arquivo:** `src/hls.js`

**Extensão de `parsePlaylistText`:**

```js
// Tag: #EXT-X-MEDIA:TYPE=AUDIO,GROUP-ID="audio",LANGUAGE="pt",NAME="Português",DEFAULT=YES,AUTOSELECT=YES,URI="audio_pt.m3u8"
// Tag: #EXT-X-MEDIA:TYPE=SUBTITLES,GROUP-ID="subs",LANGUAGE="en",NAME="English",DEFAULT=NO,URI="sub_en.vtt"

function parsePlaylistText(text, baseUrl) {
  const lines = text.split(/\r?\n/);
  const variants = [];
  const audioTracks = [];    // NOVO
  const subtitleTracks = []; // NOVO
  let baseUrl = baseUrl;

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i].trim();

    // Parse #EXT-X-MEDIA (áudio e legendas)
    if (line.startsWith('#EXT-X-MEDIA:')) {
      const attrs = parseAttributes(line.slice('#EXT-X-MEDIA:'.length));
      const type = attrs.TYPE;
      if (type === 'AUDIO') {
        audioTracks.push({
          language: attrs.LANGUAGE || 'und',
          label: attrs.NAME || '',
          isDefault: attrs.DEFAULT === 'YES',
          uri: attrs.URI ? resolveUri(attrs.URI, baseUrl) : null,
          groupId: attrs['GROUP-ID'] || '',
        });
      } else if (type === 'SUBTITLES') {
        subtitleTracks.push({
          language: attrs.LANGUAGE || 'und',
          label: attrs.NAME || '',
          isDefault: attrs.DEFAULT === 'YES',
          uri: attrs.URI ? resolveUri(attrs.URI, baseUrl) : null,
          groupId: attrs['GROUP-ID'] || '',
        });
      }
      continue;
    }

    // Parse #EXT-X-STREAM-INF (existente, mas adicionar AUDIO/SUBTITLES refs)
    if (line.startsWith('#EXT-X-STREAM-INF:')) {
      // ... parse existente ...
      // NOVO: extrair AUDIO e SUBTITLES attributes
      const audioGroup = attrs.AUDIO || '';
      const subsGroup = attrs.SUBTITLES || '';
      // Vincular ao variant...
    }
  }

  return { kind, variants, audioTracks, subtitleTracks, baseUrl };
}
```

### 4.2 Helper `parseAttributes`

```js
/**
 * Parse HLS attribute list: KEY1=value1,KEY2="value with spaces"
 */
function parseAttributes(str) {
  const attrs = {};
  const re = /([A-Z0-9-]+)=(?:"([^"]*)"|([^,]*))/g;
  let m;
  while ((m = re.exec(str))) {
    attrs[m[1]] = m[2] ?? m[3];
  }
  return attrs;
}
```

### 6.2 Nova função `runMuxMultiDownload`

```js
/**
 * Baixa vídeo + N faixas de áudio e faz mux com FFmpeg.
 * Cada faixa de áudio vira um input separado no FFmpeg.
 */
async function runMuxMultiDownload(prepared, output, headers, signal, onProgress) {
  const tmpDir = fs.mkdtempSync(path.join(os.tmpdir(), 'sg-mux-multi-'));
  try {
    // 1) Baixar vídeo
    const videoTmp = path.join(tmpDir, 'video.mp4');
    const videoResult = await runStreamDownload(
      prepared.videoUrl, videoTmp, headers, signal,
      (u) => onProgress({ ...u, stage: 'downloading', message: 'Baixando vídeo' })
    );
    if (!videoResult.ok) return videoResult;
    if (signal?.aborted) return abortOutcome(signal);

    // 2) Baixar cada faixa de áudio em paralelo
    const audioResults = [];
    for (let i = 0; i < prepared.audioUrls.length; i++) {
      const audioTmp = path.join(tmpDir, `audio_${i}.m4a`);
      const result = await runStreamDownload(
        prepared.audioUrls[i], audioTmp, headers, signal,
        (u) => onProgress({
          ...u,
          stage: 'downloading',
          message: `Baixando áudio ${i + 1}/${prepared.audioUrls.length} (${prepared.audioLabels[i] || ''})`,
        })
      );
      if (!result.ok) return result;
      if (signal?.aborted) return abortOutcome(signal);
      audioResults.push(audioTmp);
    }

    // 3) Mux com FFmpeg: -i video -i audio0 -i audio1 ... -map 0:v -map 1:a -map 2:a ...
    onProgress?.({ stage: 'merging', percent: 90, message: 'Juntando vídeo + áudios' });

    const ffmpegArgs = ['-i', videoTmp];
    const maps = ['-map', '0:v'];
    const metadata = [];

    for (let i = 0; i < audioResults.length; i++) {
      ffmpegArgs.push('-i', audioResults[i]);
      maps.push('-map', `${i + 1}:a`);

      const lang = prepared.audioLanguages[i] || 'und';
      const label = prepared.audioLabels[i] || '';
      metadata.push('-metadata:s:a:' + i, `language=${lang}`);
      if (label) metadata.push('-metadata:s:a:' + i, `title=${label}`);
    }

    const { promise, stop } = startMuxDownload({
      videoInput: videoTmp,
      audioInputs: audioResults, // Nova prop: aceita array
      output,
      extraArgs: [...maps, ...metadata, '-c', 'copy'],
      onProgress: makeFfmpegProgress((u) => onProgress({ ...u, stage: 'merging' }), 0),
    });

    const onAbort = () => stop();
    signal?.addEventListener('abort', onAbort, { once: true });
    try {
      const result = await promise;
      if (signal?.aborted) return abortOutcome(signal);
      return result.ok ? { ok: true } : {
        ok: false, code: 'MUX_MULTI_FAILED',
        error: `ffmpeg mux saiu com codigo ${result.code}`,
      };
    } finally {
      signal?.removeEventListener('abort', onAbort);
    }
  } finally {
    fs.rmSync(tmpDir, { recursive: true, force: true });
  }
}
```

## 11. Riscos e Mitigações

| Risco | Impacto | Mitigação |
|-------|---------|-----------|
| yt-dlp não retorna `language` para todos os formatos | Alto | Fallback: usar `format_note` ou `'und'` |
| HLS sem `#EXT-X-MEDIA` (playlists simples) | Baixo | `audioTracks: []` — comportamento atual preservado |
| Mux multi-áudio aumenta tempo de download | Médio | Downloads de áudio em paralelo (Promise.all) |
| Legendas embutidas não funcionam em todos os players | Baixo | Opção off por padrão; documentar |
| Áudio `language: 'und'` ambíguo | Médio | UI mostra "Desconhecido"; CLI aceita `--audio-lang und` |
|yt-dlp flags de legenda mudam entre versões | Baixo | Usar `--convert-subs srt` para normalizar |
